117| def api_get(url):
118|     try:
119|         req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
120|         with urllib.request.urlopen(req, timeout=10) as resp:
121|             return json.loads(resp.read().decode())
122|     except Exception as e:
123|         return {"error": str(e)}

126| def get_prayer_times(lat=DEFAULT_LAT, lon=DEFAULT_LON):
127|     now_ist = datetime.now(IST)
128|     date_str = now_ist.strftime("%d-%m-%Y")
129|     url = f"{ALADHAN_BASE}/timings/{date_str}?latitude={lat}&longitude={lon}&method={CALCULATION_METHOD}"
130|     data = api_get(url)
131|     if "error" in data:
132|         return None
133|     try:
134|         timings = data["data"]["timings"]
135|         hijri = data["data"]["date"]["hijri"]
136|         return {
137|             "fajr": timings["Fajr"].split()[0],
138|             "sunrise": timings["Sunrise"].split()[0],
139|             "dhuhr": timings["Dhuhr"].split()[0],
140|             "asr": timings["Asr"].split()[0],

173| def draw_gradient(img, top_color, bottom_color):
174|     draw = ImageDraw.Draw(img)
175|     w, h = img.size
176|     for y in range(h):
177|         ratio = y / h
178|         r = int(top_color[0] + (bottom_color[0] - top_color[0]) * ratio)
179|         g = int(top_color[1] + (bottom_color[1] - top_color[1]) * ratio)
180|         b = int(top_color[2] + (bottom_color[2] - top_color[2]) * ratio)
181|         draw.line([(0, y), (w, y)], fill=(r, g, b))

190| def draw_text_with_shadow(draw, pos, text, font, fill, shadow_color=(0,0,0), offset=3):
191|     """Draw text with a shadow for better readability."""

199| def draw_centered_text(draw, y, text, font, fill, w, shadow=True):
200|     """Draw horizontally centered text. Returns new y position."""
201|     bbox = draw.textbbox((0, 0), text, font=font)
202|     tw = bbox[2] - bbox[0]
203|     th = bbox[3] - bbox[1]
204|     x = (w - tw) // 2
205|     if shadow:
206|         draw_text_with_shadow(draw, (x, y), text, font, fill)
207|     else:
208|         draw.text((x, y), text, font=font, fill=fill)
209|     return y + th + 4

212| def wrap_text(text, font, max_width, draw):
213|     """Wrap text to fit within max_width."""
214|     words = text.split()
215|     lines = []
216|     current_line = ""
217|     for word in words:
218|         test = f"{current_line} {word}".strip()
219|         bbox = draw.textbbox((0, 0), test, font=font)
220|         if bbox[2] - bbox[0] <= max_width:
221|             current_line = test
222|         else:
223|             if current_line:
224|                 lines.append(current_line)
225|             current_line = word
226|     if current_line:
227|         lines.append(current_line)
228|     return lines
