106| def api_get(url):
107|     """Make an API request and return JSON."""
108|     try:
109|         req = urllib.request.Request(url, headers={"User-Agent": "Mozilla/5.0"})
110|         with urllib.request.urlopen(req, timeout=10) as resp:
111|             return json.loads(resp.read().decode())
112|     except Exception as e:
113|         return {"error": str(e)}

116| def get_prayer_times(lat=DEFAULT_LAT, lon=DEFAULT_LON):
117|     """Fetch prayer times from AlAdhan API."""
118|     now_ist = datetime.now(IST)
119|     date_str = now_ist.strftime("%d-%m-%Y")
120|     url = f"{ALADHAN_BASE}/timings/{date_str}?latitude={lat}&longitude={lon}&method={CALCULATION_METHOD}"
121|     data = api_get(url)
122|     if "error" in data:
123|         return None
124|     try:
125|         timings = data["data"]["timings"]
126|         hijri = data["data"]["date"]["hijri"]
127|         return {
128|             "fajr": timings["Fajr"].split()[0],
129|             "sunrise": timings["Sunrise"].split()[0],
130|             "dhuhr": timings["Dhuhr"].split()[0],
131|             "asr": timings["Asr"].split()[0],
132|             "maghrib": timings["Maghrib"].split()[0],
133|             "isha": timings["Isha"].split()[0],
134|             "hijri_date": "{} {} {} AH".format(hijri['day'], hijri['month']['en'], hijri['year']),
135|             "hijri_ar": "{} {} {} \u0647\u0640".format(hijri['day'], hijri['month']['ar'], hijri['year']),

183| def draw_gradient(img, top_color, bottom_color):
184|     """Draw a vertical gradient on the image."""
185|     draw = ImageDraw.Draw(img)
186|     w, h = img.size
187|     for y in range(h):
188|         ratio = y / h
189|         r = int(top_color[0] + (bottom_color[0] - top_color[0]) * ratio)
190|         g = int(top_color[1] + (bottom_color[1] - top_color[1]) * ratio)
191|         b = int(top_color[2] + (bottom_color[2] - top_color[2]) * ratio)
192|         draw.line([(0, y), (w, y)], fill=(r, g, b))

195| def draw_decorative_border(draw, w, h, accent, padding=20):
196|     """Draw decorative Islamic-style border."""
197|     # Outer border
198|     draw.rectangle([padding, padding, w - padding, h - padding], outline=accent, width=2)
199|     # Inner border
200|     draw.rectangle([padding + 6, padding + 6, w - padding - 6, h - padding - 6], outline=accent, width=1)
201|     # Corner decorations
202|     corner_size = 15
203|     for x, y in [(padding, padding), (w - padding, padding), (padding, h - padding), (w - padding, h - padding)]:
204|         draw.rectangle([x - 3, y - 3, x + corner_size, y + corner_size], fill=accent)

207| def draw_centered_text(draw, y, text, font, fill, w, max_width=None):
208|     """Draw horizontally centered text. Returns new y position."""
209|     bbox = draw.textbbox((0, 0), text, font=font)
210|     tw = bbox[2] - bbox[0]
211|     x = (w - tw) // 2
212|     draw.text((x, y), text, font=font, fill=fill)
213|     return y + (bbox[3] - bbox[1]) + 8

216| def wrap_text(text, font, max_width, draw):
217|     """Wrap text to fit within max_width."""
218|     words = text.split()
219|     lines = []
220|     current_line = ""
221|     for word in words:
222|         test = f"{current_line} {word}".strip()
223|         bbox = draw.textbbox((0, 0), test, font=font)
224|         if bbox[2] - bbox[0] <= max_width:
225|             current_line = test
226|         else:
227|             if current_line:
228|                 lines.append(current_line)
229|             current_line = word
230|     if current_line:
231|         lines.append(current_line)
232|     return lines
